Invite staff through Authentication > Users. -- All authenticated users can read and edit the shared cost dataset. create table if not exists public.bar_data ( id integer primary key check (id = 1), payload jsonb not null, updated_at timestamptz not null default now(), updated_by uuid references auth.users(id) ); alter table public.bar_data enable row level security; revoke all on public.bar_data from anon; grant select, insert, update on public.bar_data to authenticated; drop policy if exists "Staff can read bar costs" on public.bar_data; create policy "Staff can read bar costs" on public.bar_data for select to authenticated using (true); drop policy if exists "Staff can insert bar costs" on public.bar_data; create policy "Staff can insert bar costs" on public.bar_data for insert to authenticated with check (true); drop policy if exists "Staff can update bar costs" on public.bar_data; create policy "Staff can update bar costs" on public.bar_data for update to authenticated using (true) with check (true); create or replace function public.set_bar_data_timestamp() returns trigger language plpgsql as $$ begin new.updated_at=now(); return new; end $$; drop trigger if exists bar_data_timestamp on public.bar_data; create trigger bar_data_timestamp before insert or update on public.bar_data for each row execute function public.set_bar_data_timestamp();
